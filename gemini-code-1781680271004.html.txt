    <script>
        let map;
        let motoMarker;
        let routeLine;
        let simInterval = null;
        let watchId = null;
        let currentStep = 0;

        // 1. RUTA DE PRUEBA (Coordenadas sintéticas de un puerto de montaña revirado)
        const mockRoute = [
            [36.7410, -5.1610], // Recta inicial
            [36.7425, -5.1610],
            [36.7440, -5.1610],
            [36.7455, -5.1595], // Curva suave derecha
            [36.7465, -5.1570], // Sigue derecha
            [36.7475, -5.1550], 
            [36.7490, -5.1550], // Recta intermedia
            [36.7505, -5.1552],
            [36.7515, -5.1575], // Curva cerrada izquierda
            [36.7520, -5.1605], // Sigue cerrándose izquierda
            [36.7523, -5.1635],
            [36.7535, -5.1645], // Estabilizando
            [36.7550, -5.1645]  // Final
        ];

        // 2. INICIALIZACIÓN DEL MAPA
        function initMap() {
            // Centrar mapa en el inicio de la ruta
            map = L.map('map', {
                zoomControl: false,
                attributionControl: false
            }).setView(mockRoute[0], 15);

            // Estilo de mapa ultraligero (CartoDB Positron Dark)
            // Este mapa oculta bosques, relieves complejos y resalta solo carreteras y poblaciones
            L.tileLayer('https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png', {
                maxZoom: 19
            }).addTo(map);

            // Pintar la ruta en el mapa (Línea de color Neón para alta visibilidad)
            routeLine = L.polyline(mockRoute, {
                color: '#00e5ff',
                weight: 5,
                opacity: 0.8
            }).addTo(map);

            // Marcador de la posición del motorista
            motoMarker = L.circleMarker(mockRoute[0], {
                radius: 10,
                color: '#ffffff',
                fillColor: '#ffb300',
                fillOpacity: 1,
                weight: 3
            }).addTo(map);
        }

        // 3. ALGORITMO: CALCULAR RUMBO (BEARING) ENTRE DOS PUNTOS
        function getBearing(p1, p2) {
            const lat1 = p1[0] * Math.PI / 180;
            const lat2 = p2[0] * Math.PI / 180;
            const dLon = (p2[1] - p1[1]) * Math.PI / 180;

            const y = Math.sin(dLon) * Math.cos(lat2);
            const x = Math.cos(lat1) * Math.sin(lat2) - Math.sin(lat1) * Math.cos(lat2) * Math.cos(dLon);
            
            let bearing = Math.atan2(y, x) * 180 / Math.PI;
            return (bearing + 360) % 360; // Normalizar entre 0 y 360 grados
        }

        // 4. ALGORITMO: PREDECIR LA PRÓXIMA CURVA
        function analizarProximaCurva(indiceActual, ruta) {
            if (indiceActual >= ruta.length - 2) {
                updateUI("↑", "Fin de ruta", "#fff");
                return;
            }

            // Segmento actual en el que está la moto (del punto A al punto B)
            const pA = ruta[indiceActual];
            const pB = ruta[indiceActual + 1];
            const rumboActual = getBearing(pA, pB);

            // Segmento futuro (del punto B al punto C) - Mirada hacia adelante
            const pC = ruta[indiceActual + 2];
            const rumboSiguiente = getBearing(pB, pC);

            // Calcular diferencia de ángulo
            let diferencia = rumboSiguiente - rumboActual;
            if (diferencia > 180) diferencia -= 360;
            if (diferencia < -180) diferencia += 360;

            // Umbrales de decisión (Grados de desviación)
            const UMBRAL_SUAVE = 15;
            const UMBRAL_CERRADA = 45;

            if (Math.abs(diferencia) < UMBRAL_SUAVE) {
                updateUI("↑", "Recta", "#00ff66");
            } else if (diferencia > 0) {
                if (diferencia > UMBRAL_CERRADA) {
                    updateUI("➔ ➔", "Curva Cerrada Derecha", "#ff3d00");
                } else {
                    updateUI("↗", "Curva Derecha", "#ffb300");
                }
            } else {
                if (diferencia < -UMBRAL_CERRADA) {
                    updateUI("⬅ ⬅", "Curva Cerrada Izquierda", "#ff3d00");
                } else {
                    updateUI("↖", "Curva Izquierda", "#ffb300");
                }
            }
        }

        // Actualizar la interfaz visual superior
        function updateUI(arrow, text, color) {
            const arrowEl = document.getElementById('arrow-display');
            const textEl = document.getElementById('instruction-text');
            
            arrowEl.innerText = arrow;
            arrowEl.style.color = color;
            textEl.innerText = text;
        }

        // 5. MOTOR DE SIMULACIÓN PARA EL TEST
        function toggleSimulation() {
            const btn = document.getElementById('btn-sim');
            
            if (simInterval) {
                clearInterval(simInterval);
                simInterval = null;
                btn.innerText = "Simular Ruta";
                btn.classList.remove('active');
            } else {
                // Detener GPS real si estuviera activo
                if(watchId) toggleGPS();
                
                currentStep = 0;
                btn.innerText = "Detener";
                btn.classList.add('active');
                
                simInterval = setInterval(() => {
                    if (currentStep < mockRoute.length) {
                        const targetPos = mockRoute[currentStep];
                        
                        // Actualizar marcador y mapa
                        motoMarker.setLatLng(targetPos);
                        map.panTo(targetPos);
                        
                        // Analizar trazado geométrico
                        analizarProximaCurva(currentStep, mockRoute);
                        
                        currentStep++;
                    } else {
                        clearInterval(simInterval);
                        simInterval = null;
                        btn.innerText = "Simular Ruta";
                        btn.classList.remove('active');
                    }
                }, 2500); // Avanza de punto cada 2.5 segundos
            }
        }

        // 6. CAPTURA DE GPS REAL (Para probar en calle con el móvil)
        function toggleGPS() {
            const btn = document.getElementById('btn-gps');
            
            if (watchId) {
                navigator.geolocation.clearWatch(watchId);
                watchId = null;
                btn.innerText = "GPS Real";
                btn.classList.remove('active');
            } else {
                if (simInterval) toggleSimulation(); // Detener simulación si activa
                
                if (!navigator.geolocation) {
                    alert("Tu navegador no soporta GPS.");
                    return;
                }
                
                btn.innerText = "Apagar GPS";
                btn.classList.add('active');
                
                watchId = navigator.geolocation.watchPosition(
                    (position) => {
                        const userPos = [position.coords.latitude, position.coords.longitude];
                        
                        motoMarker.setLatLng(userPos);
                        map.panTo(userPos);
                        
                        // NOTA: Para que funcione el predictor con GPS real en calle, 
                        // deberías vincular la posición actual al nodo más cercano de un archivo GPX cargado.
                        updateUI("📡", "GPS Activo - Localizando", "#00e5ff");
                    },
                    (err) => {
                        console.error(err);
                        alert("Error al obtener posición GPS.");
                        toggleGPS();
                    },
                    {
                        enableHighAccuracy: true, // Forzar uso de hardware GPS integrado
                        maximumAge: 0,
                        timeout: 5000
                    }
                );
            }
        }

        // Arrancar la app al cargar la ventana
        window.onload = initMap;
    </script>
