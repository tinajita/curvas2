    <script src="https://unpkg.com/maplibre-gl@4.3.2/dist/maplibre-gl.js"></script>

    <script>
        let map;
        let arrowMarker;
        let miPosicion = [-4.4214, 36.7213]; // Málaga inicial
        let lastValidBearing = 0;
        
        let mapaDesplazadoPorUsuario = false;
        let autoRecenterTimer = null;

        const MAP_STYLE = 'https://basemaps.cartocdn.com/gl/dark-matter-gl-style/style.json';

        // CLAVE: Calculamos el píxel exacto del desfase en el eje Y.
        // Queremos que el punto del mapa se dibuje un 22% por debajo del centro real de la pantalla.
        const getCenterOffset = () => [0, window.innerHeight * 0.22];

        function initNavigator() {
            map = new maplibregl.Map({
                container: 'map',
                style: MAP_STYLE,
                center: miPosicion, 
                zoom: 16.5,                 
                pitch: 60, // Ángulo de perspectiva pronunciado                 
                bearing: 0,
                attributionControl: false
            });

            map.on('load', () => {
                configurarCapasRuta();
                crearIconoNavegacion();
                arrancarSeguimientoGPS();
                configurarEventosGestos();
                
                // Aplicamos el desfase inicial
                setTimeout(() => { recenterManual(); }, 300);
            });
        }

        function configurarEventosGestos() {
            map.on('dragstart', () => {
                mapaDesplazadoPorUsuario = true;
                document.getElementById('recenter-container').style.display = 'block';
                clearTimeout(autoRecenterTimer);
            });

            map.on('dragend', () => {
                arrancarTemporizadorRecentrando();
            });
        }

        function arrancarTemporizadorRecentrando() {
            clearTimeout(autoRecenterTimer);
            autoRecenterTimer = setTimeout(() => {
                recenterManual();
            }, 6000);
        }

        function recenterManual() {
            mapaDesplazadoPorUsuario = false;
            document.getElementById('recenter-container').style.display = 'none';
            clearTimeout(autoRecenterTimer);
            
            // Usamos 'offset' para desplazar el eje de la cámara hacia abajo de forma limpia
            map.easeTo({
                center: miPosicion,
                zoom: 16.5,
                bearing: lastValidBearing,
                pitch: 60,
                offset: getCenterOffset(),
                duration: 600
            });
        }

        function configurarCapasRuta() {
            map.addSource('route-source', {
                type: 'geojson',
                data: { type: 'FeatureCollection', features: [] }
            });

            map.addLayer({
                id: 'route-casing',
                type: 'line',
                source: 'route-source',
                layout: { 'line-join': 'round', 'line-cap': 'round' },
                paint: { 'line-color': '#000000', 'line-width': 16, 'line-opacity': 0.85 }
            });

            map.addLayer({
                id: 'route-core',
                type: 'line',
                source: 'route-source',
                layout: { 'line-join': 'round', 'line-cap': 'round' },
                paint: { 'line-color': '#00ff66', 'line-width': 8, 'line-opacity': 0.95 }
            });
        }

        function crearIconoNavegacion() {
            const container = document.createElement('div');
            container.className = 'nav-arrow-marker';
            container.innerHTML = `
                <svg width="42" height="42" viewBox="0 0 100 100" fill="none" xmlns="http://www.w3.org/2000/svg">
                    <path d="M50 5L90 90L50 70L10 90L50 5Z" fill="#ffffff" stroke="#00e5ff" stroke-width="6" stroke-linejoin="round"/>
                </svg>
            `;
            arrowMarker = new maplibregl.Marker({ element: container }).setLngLat(miPosicion).addTo(map);
        }

        async function fijarRutaDestino() {
            const query = document.getElementById('destination-input').value;
            if (!query) return;

            try {
                const searchResponse = await fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(query)}&limit=1`);
                const searchData = await searchResponse.json();
                if (searchData.length === 0) return;

                const destLon = parseFloat(searchData[0].lon);
                const destLat = parseFloat(searchData[0].lat);

                const routeResponse = await fetch(`https://router.project-osrm.org/route/v1/driving/${miPosicion[0]},${miPosicion[1]};${destLon},${destLat}?overview=full&geometries=geojson`);
                const routeData = await routeResponse.json();
                if (!routeData.routes || routeData.routes.length === 0) return;

                document.getElementById('search-block').style.display = 'none';
                const pulgaTop = document.getElementById('nav-pill-top');
                pulgaTop.style.display = 'flex';
                pulgaTop.querySelector('#current-road').innerText = query.split(',')[0];

                map.getSource('route-source').setData({
                    type: 'FeatureCollection',
                    features: [{ type: 'Feature', geometry: routeData.routes[0].geometry, properties: {} }]
                });

                recenterManual();

            } catch (error) {
                console.error(error);
            }
        }

        function cancelarRuta() {
            map.getSource('route-source').setData({ type: 'FeatureCollection', features: [] });
            document.getElementById('nav-pill-top').style.display = 'none';
            document.getElementById('search-block').style.display = 'flex';
            document.getElementById('destination-input').value = '';
            recenterManual();
        }

        function arrancarSeguimientoGPS() {
            if (!navigator.geolocation) return;

            navigator.geolocation.watchPosition(
                (position) => {
                    const lon = position.coords.longitude;
                    const lat = position.coords.latitude;
                    const velocidad = position.coords.speed ? Math.round(position.coords.speed * 3.6) : 0;
                    
                    miPosicion = [lon, lat];

                    if (position.coords.heading !== null && position.coords.heading !== undefined) {
                        lastValidBearing = position.coords.heading;
                    }

                    document.getElementById('speed-value').innerText = velocidad;
                    document.getElementById('gps-status').innerText = `SAT OK (±${Math.round(position.coords.accuracy)}m)`;

                    // Sincronizamos la flecha exactamente con la posición actual
                    arrowMarker.setLngLat(miPosicion);

                    const arrowElement = document.querySelector('.nav-arrow-marker');
                    if (arrowElement) {
                        arrowElement.style.transform = `rotate(${lastValidBearing}deg)`;
                    }

                    // En movimiento constante, aplicamos el desfase (offset) para mantener la flecha abajo
                    if (!mapaDesplazadoPorUsuario) {
                        map.easeTo({
                            center: miPosicion,
                            bearing: lastValidBearing,
                            offset: getCenterOffset(),
                            duration: 250,
                            easing: (t) => t
                        });
                    }
                },
                (err) => console.error(err),
                { enableHighAccuracy: true, maximumAge: 0, timeout: 5000 }
            );
        }

        window.onload = initNavigator;
    </script>
