    <script src="https://unpkg.com/maplibre-gl@4.3.2/dist/maplibre-gl.js"></script>

    <script>
        let map;
        let arrowMarker;
        let miPosicion = [-4.4214, 36.7213]; // Coordenadas Málaga base
        let destinoCoors = null;
        let lastValidBearing = 0;
        
        let mapaDesplazadoPorUsuario = false;
        let autoRecenterTimer = null;
        let rutaActiva = false;

        const MAP_STYLE = 'https://basemaps.cartocdn.com/gl/dark-matter-gl-style/style.json';
        const getCenterOffset = () => [0, window.innerHeight * 0.22]; // Clava la flecha abajo

        function initNavigator() {
            map = new maplibregl.Map({
                container: 'map',
                style: MAP_STYLE,
                center: miPosicion, 
                zoom: 16.5,                 
                pitch: 60,                 
                bearing: 0,
                attributionControl: false
            });

            map.on('load', () => {
                configurarCapasRuta();
                crearIconoNavegacion();
                arrancarSeguimientoGPS();
                configurarEventosGestos();
                setTimeout(() => { recenterManual(); }, 300);
            });
        }

        function configurarEventosGestos() {
            map.on('dragstart', () => {
                mapaDesplazadoPorUsuario = true;
                document.getElementById('recenter-container').style.display = 'block';
                clearTimeout(autoRecenterTimer);
            });
            map.on('dragend', () => { arrancarTemporizadorRecentrando(); });
        }

        function arrancarTemporizadorRecentrando() {
            clearTimeout(autoRecenterTimer);
            autoRecenterTimer = setTimeout(() => { recenterManual(); }, 6000);
        }

        function recenterManual() {
            mapaDesplazadoPorUsuario = false;
            document.getElementById('recenter-container').style.display = 'none';
            clearTimeout(autoRecenterTimer);
            
            map.easeTo({
                center: miPosicion,
                zoom: 16.5,
                bearing: lastValidBearing,
                pitch: 60,
                offset: getCenterOffset(),
                duration: 600
            });
        }

        function configurarCapasRuta() {
            map.addSource('route-source', {
                type: 'geojson',
                data: { type: 'FeatureCollection', features: [] }
            });

            map.addLayer({
                id: 'route-casing',
                type: 'line',
                source: 'route-source',
                layout: { 'line-join': 'round', 'line-cap': 'round' },
                paint: { 'line-color': '#000000', 'line-width': 16, 'line-opacity': 0.85 }
            });

            map.addLayer({
                id: 'route-core',
                type: 'line',
                source: 'route-source',
                layout: { 'line-join': 'round', 'line-cap': 'round' },
                paint: { 'line-color': '#00ff66', 'line-width': 8, 'line-opacity': 0.95 }
            });
        }

        function crearIconoNavegacion() {
            const container = document.createElement('div');
            container.className = 'nav-arrow-marker';
            container.innerHTML = `
                <svg width="42" height="42" viewBox="0 0 100 100" fill="none" xmlns="http://www.w3.org/2000/svg">
                    <path d="M50 5L90 90L50 70L10 90L50 5Z" fill="#ffffff" stroke="#00e5ff" stroke-width="6" stroke-linejoin="round"/>
                </svg>
            `;
            arrowMarker = new maplibregl.Marker({ element: container }).setLngLat(miPosicion).addTo(map);
        }

        // --- CALCULAR E INICIAR MOTOR DE RUTA ---
        async function fijarRutaDestino() {
            const query = document.getElementById('destination-input').value;
            if (!query) return;

            try {
                const searchResponse = await fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(query)}&limit=1`);
                const searchData = await searchResponse.json();
                if (searchData.length === 0) return;

                destinoCoors = [parseFloat(searchData[0].lon), parseFloat(searchData[0].lat)];
                rutaActiva = true;

                // Transición estética de UI
                document.getElementById('search-block').style.display = 'none';
                document.getElementById('nav-pill-top').style.display = 'flex';
                document.getElementById('route-stats-pill').style.display = 'flex';

                // Llamada inicial para pintar e info de telemetría
                actualizarRutaEnMarcha();
                recenterManual();

            } catch (error) {
                console.error(error);
            }
        }

        // Refresca la línea y los datos de distancia/tiempo usando la posición actual real
        async function actualizarRutaEnMarcha() {
            if (!rutaActiva || !destinoCoors) return;

            try {
                const routeResponse = await fetch(`https://router.project-osrm.org/route/v1/driving/${miPosicion[0]},${miPosicion[1]};${destinoCoors[0]},${destinoCoors[1]}?overview=full&geometries=geojson`);
                const routeData = await routeResponse.json();
                
                if (!routeData.routes || routeData.routes.length === 0) return;

                const dataRuta = routeData.routes[0];

                // 1. Actualizar trazo neón en el mapa
                map.getSource('route-source').setData({
                    type: 'FeatureCollection',
                    features: [{ type: 'Feature', geometry: dataRuta.geometry, properties: {} }]
                });

                // 2. Procesar y formatear Distancia (Metros -> Kilómetros)
                const kms = (dataRuta.distance / 1000).toFixed(1);
                document.getElementById('stat-distance').innerHTML = `<span>${kms}</span> KM`;

                // 3. Procesar y formatear Duración (Segundos -> Minutos / Horas)
                const totalMinutos = Math.round(dataRuta.duration / 60);
                if (totalMinutos >= 60) {
                    const horas = Math.floor(totalMinutos / 60);
                    const minsRestantes = totalMinutos % 60;
                    document.getElementById('stat-time').innerHTML = `<span>${horas}h ${minsRestantes}m</span>`;
                } else {
                    document.getElementById('stat-time').innerHTML = `<span>${totalMinutos}</span> MIN`;
                }

            } catch (e) {
                console.error("Error actualizando telemetría de ruta:", e);
            }
        }

        function cancelarRuta() {
            rutaActiva = false;
            destinoCoors = null;
            
            map.getSource('route-source').setData({ type: 'FeatureCollection', features: [] });
            
            document.getElementById('nav-pill-top').style.display = 'none';
            document.getElementById('route-stats-pill').style.display = 'none';
            document.getElementById('search-block').style.display = 'flex';
            document.getElementById('destination-input').value = '';
            
            recenterManual();
        }

        // --- DETECTOR DINÁMICO DE CURVAS / CARRETERA ACTUAL (VÍA) ---
        function identificarYMostrarViaActual(lon, lat) {
            const pixelPos = map.project([lon, lat]);
            const bbox = [
                [pixelPos.x - 12, pixelPos.y - 12],
                [pixelPos.x + 12, pixelPos.y + 12]
            ];

            const features = map.queryRenderedFeatures(bbox);
            const road = features.find(f => 
                f.layer.id.includes('road') || 
                f.layer.id.includes('highway') ||
                f.properties.class === 'highway'
            );

            const textoPill = document.getElementById('current-road');
            if (road && road.properties) {
                const nombreVía = road.properties.name || road.properties.ref || "En ruta";
                textoPill.innerText = nombreVía;
            } else if (!rutaActiva) {
                textoPill.innerText = "Localizando Vía...";
            }
        }

        // --- BUCLE DE SEGUIMIENTO GPS HARDWARE ---
        function arrancarSeguimientoGPS() {
            if (!navigator.geolocation) return;

            navigator.geolocation.watchPosition(
                (position) => {
                    const lon = position.coords.longitude;
                    const lat = position.coords.latitude;
                    const velocidad = position.coords.speed ? Math.round(position.coords.speed * 3.6) : 0;
                    
                    miPosicion = [lon, lat];

                    if (position.coords.heading !== null && position.coords.heading !== undefined) {
                        lastValidBearing = position.coords.heading;
                    }

                    // Renderizar velocímetro y precisión
                    document.getElementById('speed-value').innerText = velocidad;
                    document.getElementById('gps-status').innerText = `Satélites OK (±${Math.round(position.coords.accuracy)}m)`;

                    // Fijar posición de flecha
                    arrowMarker.setLngLat(miPosicion);

                    // Rotar flecha
                    const arrowElement = document.querySelector('.nav-arrow-marker');
                    if (arrowElement) {
                        arrowElement.style.transform = `rotate(${lastValidBearing}deg)`;
                    }

                    // Si hay una ruta activa, recalculamos consumos de distancia y trazo restante
                    if (rutaActiva) {
                        actualizarRutaEnMarcha();
                    }

                    // Leer baldosas vectoriales para sacar el nombre de las curvas
                    identificarYMostrarViaActual(lon, lat);

                    // Mover mapa con el offset estricto abajo si el usuario no está tocando la pantalla
                    if (!mapaDesplazadoPorUsuario) {
                        map.easeTo({
                            center: miPosicion,
                            bearing: lastValidBearing,
                            offset: getCenterOffset(),
                            duration: 250,
                            easing: (t) => t
                        });
                    }
                },
                (err) => console.error(err),
                { enableHighAccuracy: true, maximumAge: 0, timeout: 5000 }
            );
        }

        window.addEventListener('resize', () => {
            if (map) map.setPadding({ top: 0, bottom: 0 }); // Evitamos conflictos de padding con el offset
        });

        window.onload = initNavigator;
    </script>
