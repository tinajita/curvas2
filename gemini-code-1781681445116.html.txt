    <script>
        let map;
        let arrowMarker;
        let lastValidBearing = 0;

        // Estilo oscuro real de CartoDB listo para renderizado vectorial por GPU
        const MAP_STYLE = 'https://basemaps.cartocdn.com/gl/dark-matter-gl-style/style.json';

        function initNavigator() {
            map = new maplibregl.Map({
                container: 'map',
                style: MAP_STYLE,
                center: [-4.4214, 36.7213], // Centro inicial por defecto
                zoom: 16.5,                 // Zoom cerrado ideal para ver curvas con antelación
                pitch: 55,                  // Inclinación pronunciada para efecto 3D de conducción
                bearing: 0,
                attributionControl: false
            });

            map.on('load', () => {
                configurarCapasResaltado();
                crearIconoNavegacion();
                arrancarSeguimientoGPS();
            });
        }

        // --- 1. CONFIGURACIÓN DEL RESALTADO DE CARRETERA REAL ---
        function configurarCapasResaltado() {
            // Añadimos una fuente de datos GeoJSON vacía que actualizaremos con la carretera actual
            map.addSource('highlighted-road-source', {
                type: 'geojson',
                data: { type: 'FeatureCollection', features: [] }
            });

            // Capa Inferior (Casing/Borde): Da contraste negro para que el neón resalte sobre cualquier fondo
            map.addLayer({
                id: 'road-highlight-casing',
                type: 'line',
                source: 'highlighted-road-source',
                layout: { 'line-join': 'round', 'line-cap': 'round' },
                paint: {
                    'line-color': '#000000',
                    'line-width': 14,
                    'line-opacity': 0.8
                }
            });

            // Capa Superior (Core/Núcleo): Línea fina de color neón brillante
            map.addLayer({
                id: 'road-highlight-core',
                type: 'line',
                source: 'highlighted-road-source',
                layout: { 'line-join': 'round', 'line-cap': 'round' },
                paint: {
                    'line-color': '#00ff66',
                    'line-width': 8,
                    'line-opacity': 0.95
                }
            });
        }

        // --- 2. DISEÑO DEL ICONO DE POSICIÓN (FLECHA DINÁMICA) ---
        function crearIconoNavegacion() {
            const container = document.createElement('div');
            container.className = 'nav-arrow-marker';

            // SVG de flecha de navegación de alta visibilidad (Blanca con borde cian de contraste)
            container.innerHTML = `
                <svg width="40" height="40" viewBox="0 0 100 100" fill="none" xmlns="http://www.w3.org/2000/svg">
                    <path d="M50 5L90 90L50 70L10 90L50 5Z" fill="#ffffff" stroke="#00e5ff" stroke-width="6" stroke-linejoin="round"/>
                </svg>
            `;

            // Inicializamos el marcador en el mapa
            arrowMarker = new maplibregl.Marker({ element: container })
                .setLngLat([0, 0])
                .addTo(map);
        }

        // --- 3. SEGUIMIENTO GPS POR HARDWARE (SIN SIMULACIONES) ---
        function arrancarSeguimientoGPS() {
            if (!navigator.geolocation) {
                document.getElementById('current-road').innerText = "Sin soporte GPS";
                return;
            }

            navigator.geolocation.watchPosition(
                (position) => {
                    const lon = position.coords.longitude;
                    const lat = position.coords.latitude;
                    const velocidad = position.coords.speed ? Math.round(position.coords.speed * 3.6) : 0;
                    
                    // Priorizar el rumbo del hardware GPS; si estás parado, mantener el último rumbo válido
                    if (position.coords.heading !== null && position.coords.heading !== undefined) {
                        lastValidBearing = position.coords.heading;
                    }

                    // Actualizar pantallas del cuadro de mandos
                    document.getElementById('speed-value').innerText = velocidad;
                    document.getElementById('gps-accuracy').innerText = `GPS: ±${Math.round(position.coords.accuracy)}m`;

                    // Mover la flecha física en el mapa
                    arrowMarker.setLngLat([lon, lat]);

                    // Rotar el icono de la flecha mediante CSS según la dirección de la marcha
                    const arrowElement = document.querySelector('.nav-arrow-marker');
                    if (arrowElement) {
                        arrowElement.style.transform = `rotate(${lastValidBearing}deg)`;
                    }

                    // Transición suave de la cámara del navegador (Sigue a la moto y rota el mapa al rumbo)
                    map.easeTo({
                        center: [lon, lat],
                        bearing: lastValidBearing,
                        duration: 400, // Ajuste milimétrico para balancear consumo de batería y fluidez
                        easing: (t) => t
                    });

                    // Escanear qué carretera pisamos en este instante
                    identificarYResaltarVía(lon, lat);
                },
                (err) => {
                    console.error(err);
                    document.getElementById('current-road').innerText = "Buscando señal cielo abierto...";
                },
                {
                    enableHighAccuracy: true, // Activa el chip GNSS dedicado del iPhone/Android
                    maximumAge: 0,
                    timeout: 4000
                }
            );
        }

        // --- 4. DETECTOR ULTRA-RÁPIDO DE CARRETERAS VECTORIALES ---
        function identificarYResaltarVía(lon, lat) {
            const pixelPos = map.project([lon, lat]);

            // Caja de muestreo estrecha (20x20px) directamente bajo el icono de la flecha
            const bbox = [
                [pixelPos.x - 10, pixelPos.y - 10],
                [pixelPos.x + 10, pixelPos.y + 10]
            ];

            const features = map.queryRenderedFeatures(bbox);

            // Filtro de precisión para localizar el vector de la carretera en OpenStreetMap
            const road = features.find(f => 
                f.layer.id.includes('road') || 
                f.layer.id.includes('highway') ||
                f.properties.class === 'highway'
            );

            if (road && road.geometry) {
                // Obtener el identificador oficial (Ej: "A-367" o "Calle Real")
                const nombreVía = road.properties.name || road.properties.ref || "Carretera Local";
                document.getElementById('current-road').innerText = nombreVía;

                // Pintar la geometría completa de ese tramo con el doble trazado neón
                map.getSource('highlighted-road-source').setData({
                    type: 'FeatureCollection',
                    features: [{
                        type: 'Feature',
                        geometry: road.geometry,
                        properties: {}
                    }]
                });
            } else {
                // Si estás en un garaje o pista forestal no cartografiada, mantiene el aviso de ruta activo
                document.getElementById('current-road').innerText = "Ruta en Moto";
            }
        }

        window.onload = initNavigator;
    </script>
