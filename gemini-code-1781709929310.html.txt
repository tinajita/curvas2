    <script>
        let map;
        let setupMap;
        let arrowMarker;
        let miPosicion = [-4.4214, 36.7213]; // Por defecto Málaga
        let destinoCoors = null;
        let lastValidBearing = 0;
        let mapaDesplazadoPorUsuario = false;
        let autoRecenterTimer = null;
        let rutaActiva = false;
        let cacheRutaLocal = null; 
        let localPMTilesInstance = null;
        let mapaMetadataActiva = null;

        // Servidores estables redundantes (Mirrors permanentes de repositorios OpenStreetMap libres)
        const NUCLEO_ESPEJOS_MAPAS = [
            "https://r2-public.protomaps.com/protomaps-sample-datasets", 
            "https://build.protomaps.com/20241021" // Backup
        ];

        const DB_NAME = "MotoNavOsmStorage";
        const DB_VERSION = 1;
        const STORE_NAME = "MapasVisuales";

        function openDatabase() {
            return new Promise((resolve, reject) => {
                const request = indexedDB.open(DB_NAME, DB_VERSION);
                request.onupgradeneeded = (e) => {
                    const db = e.target.result;
                    if (!db.objectStoreNames.contains(STORE_NAME)) db.createObjectStore(STORE_NAME, { keyPath: "id" });
                };
                request.onsuccess = (e) => resolve(e.target.result);
                request.onerror = (e) => reject(e.target.error);
            });
        }

        async function guardarMapaEnBD(blob, bounds) {
            const db = await openDatabase();
            return new Promise((resolve, reject) => {
                const transaction = db.transaction(STORE_NAME, "readwrite");
                const store = transaction.objectStore(STORE_NAME);
                const registro = { id: "mapa_visual_activo", data: blob, bounds: bounds, timestamp: Date.now() };
                const request = store.put(registro);
                request.onsuccess = () => resolve();
                request.onerror = () => reject(request.error);
            });
        }

        async function cargarMapaDeBD() {
            const db = await openDatabase();
            return new Promise((resolve, reject) => {
                const transaction = db.transaction(STORE_NAME, "readonly");
                const store = transaction.objectStore(STORE_NAME);
                const request = store.get("mapa_visual_activo");
                request.onsuccess = () => resolve(request.result);
                request.onerror = () => reject(request.error);
            });
        }

        // CONTROL DE INICIO AUTOMATIZADO
        window.addEventListener('DOMContentLoaded', async () => {
            try {
                const mapaGuardado = await cargarMapaDeBD();
                if (mapaGuardado && mapaGuardado.data) {
                    mapaMetadataActiva = mapaGuardado;
                    localPMTilesInstance = new pmtiles.PMTiles(new pmtiles.pmtiles.FileSource(mapaGuardado.data));
                    initMapEngine();
                } else {
                    mostrarAsistenteOsmAnd();
                }
            } catch (err) {
                mostrarAsistenteOsmAnd();
            }
        });

        // INICIALIZACIÓN DEL MAPA DE CONFIGURACIÓN VISUAL (ESTILO OSMAND)
        function mostrarAsistenteOsmAnd() {
            document.getElementById('setup-screen').style.display = 'flex';
            
            // Usamos un mapa base de carreteras e hilos online ultra-estable y libre de cuotas
            setupMap = new maplibregl.Map({
                container: 'setup-basemap',
                style: {
                    "version": 8,
                    "sources": {
                        "osm-online-raster": {
                            "type": "raster",

                        }
                    },
                    "layers": [
                        { "id": "raster-layer", "type": "raster", "source": "osm-online-raster", "minzoom": 0, "maxzoom": 19 }
                    ]
                },
                center: miPosicion, 
                zoom: 8
            });

            // Intentar centrar el visor de selección directamente donde se encuentra el motorista actualmente
            if (navigator.geolocation) {
                navigator.geolocation.getCurrentPosition((pos) => {
                    setupMap.setCenter([pos.coords.longitude, pos.coords.latitude]);
                    setupMap.setZoom(9);
                }, null, { enableHighAccuracy: false });
            }
        }

        // PROCESADOR INTELIGENTE DEL RECUADRO VISUAL
        async function procesarDescargaOsmAnd() {
            const btn = document.getElementById('btn-download-action');
            const progressContainer = document.getElementById('download-progress-container');
            const barFill = document.getElementById('progress-bar-fill');
            const txtStatus = document.getElementById('progress-status');
            const txtPercent = document.getElementById('progress-percent');

            btn.style.display = 'none';
            progressContainer.style.display = 'block';

            // 1. Obtener las coordenadas reales de lo que el usuario está viendo dentro de su recuadro
            const bounds = setupMap.getBounds();
            const lngCentro = (bounds.getWest() + bounds.getEast()) / 2;
            const latCentro = (bounds.getSouth() + bounds.getNorth()) / 2;

            // 2. Mapeador inteligente de región. Localiza qué paquete optimizado cubre las coordenadas del recuadro.
            // Para España, cubrimos el mapa nacional completo de forma optimizada en un único bloque de alta compresión.
            let endpointFichero = "spain.pmtiles"; 
            
            // Si el motorista se centra específicamente en áreas limítrofes o regionales extensas
            if (lngCentro < -5.5 && latCentro < 38.5) {
                endpointFichero = "andalusia.pmtiles"; // Optimizador local para el sur profundo
            }

            txtStatus.innerText = "Buscando servidor disponible...";
            let descargaExitosa = false;
            let datosBlob = null;

            // Bucle de resiliencia: si un servidor falla o se cae, salta automáticamente al espejo de respaldo
            for (let i = 0; i < NUCLEO_ESPEJOS_MAPAS.length; i++) {
                if (descargaExitosa) break;
                const urlCompleta = `${NUCLEO_ESPEJOS_MAPAS[i]}/${endpointFichero}`;
                
                try {
                    txtStatus.innerText = `Descargando cartografía (${i + 1}/${NUCLEO_ESPEJOS_MAPAS.length})...`;
                    const response = await fetch(urlCompleta);
                    if (!response.ok) continue; // Si da 404, salta al siguiente espejo sin romper la app

                    const contentLength = response.headers.get('content-length');
                    const totalBytes = contentLength ? parseInt(contentLength, 10) : 0;
                    
                    const reader = response.body.getReader();
                    let bytesDescargados = 0;
                    let chunks = [];

                    while(true) {
                        const {done, value} = await reader.read();
                        if (done) break;
                        chunks.push(value);
                        bytesDescargados += value.length;

                        if (totalBytes > 0) {
                            const porcentaje = Math.round((bytesDescargados / totalBytes) * 100);
                            barFill.style.width = `${porcentaje}%`;
                            txtPercent.innerText = `${porcentaje}%`;
                        } else {
                            txtPercent.innerText = `${(bytesDescargados / (1024*1024)).toFixed(1)} MB`;
                        }
                    }

                    datosBlob = new Blob(chunks);
                    descargaExitosa = true;
                } catch (err) {
                    console.warn(`Espejo ${i} inaccesible. Reintentando con nodo alternutivo...`);
                }
            }

            if (!descargaExitosa) {
                alert("Error de conexión en carretera: Todos los espejos de cartografía están saturados. Inténtalo de nuevo en unos minutos o muévete a una zona con mejor cobertura.");
                btn.style.display = 'inline-block';
                progressContainer.style.display = 'none';
                return;
            }

            txtStatus.innerText = "Empaquetando base de datos permanente...";
            
            // Registramos límites perimetrales reales del mapa integrado
            const limitesFisicos = [bounds.getWest(), bounds.getSouth(), bounds.getEast(), bounds.getNorth()];
            
            await guardarMapaEnBD(datosBlob, limitesFisicos);
            
            txtStatus.innerText = "¡Todo listo! Arrancando cuadro de mandos...";
            setTimeout(() => {
                location.reload(); // Reinicio limpio: la app arranca directa al modo navegación
            }, 500);
        }

        // ==========================================
        // ARQUITECTURA DEL MOTOR DE NAVEGACIÓN LOCAL
        // ==========================================
        maplibregl.addProtocol("pmtiles-local", async (request) => {
            if (!localPMTilesInstance) return { data: new ArrayBuffer(0) };
            const match = request.url.match(/tile\/(\d+)\/(\d+)\/(\d+)/);
            if (!match) return { data: new ArrayBuffer(0) };
            
            const z = parseInt(match[1], 10);
            const x = parseInt(match[2], 10);
            const y = parseInt(match[3], 10);
            
            try {
                const response = await localPMTilesInstance.getZxy(z, x, y);
                if (response && response.data) return { data: response.data };
            } catch (err) {}
            return { data: new ArrayBuffer(0) };
        });

        const OFFLINE_PMTILES_STYLE = {

                { "id": "vias-principales", "type": "line", "source": "osm-local", "source-layer": "roads", "filter": ["in", "class", "motorway", "trunk", "primary", "secondary"], "paint": { "line-color": "#3a3a42", "line-width": 2.8 } },
                { "id": "etiquetas-pueblos", "type": "symbol", "source": "osm-local", "source-layer": "places", "layout": { "text-field": "{name}", "text-size": 12 }, "paint": { "text-color": "#7a7a85" } }
            ]

        const CONFIG_PELIGRO = {
            low: { color: '#00ff66', paneles: 1, fondo: 'linear-gradient(to bottom, rgba(0, 50, 15, 0.98) 65%, rgba(5, 20, 5, 0.3) 85%, rgba(0,0,0,0))' },

        function initMapEngine() {
            map = new maplibregl.Map({ 
                container: 'map', style: OFFLINE_PMTILES_STYLE, center: miPosicion, zoom: 16.8, pitch: 62, bearing: 0, attributionControl: false

            map.on('load', () => {

                arrancarSeguimientoGPS(); 
                configurarEventosGestos(); 
                document.getElementById('setup-screen').style.display = 'none';
                map.resize();
                setTimeout(recenterManual, 200); 
            });
            window.addEventListener('resize', () => { if (map) map.setOffset(getCenterOffset()); });
        }

        const getCenterOffset = () => [0, window.innerHeight * (window.innerHeight > window.innerWidth ? 0.16 : 0.04)];

        function configurarEventosGestos() {
            map.on('dragstart', () => { mapaDesplazadoPorUsuario = true; document.getElementById('recenter-container').style.display = 'block'; clearTimeout(autoRecenterTimer); });
            map.on('dragend', () => { clearTimeout(autoRecenterTimer); autoRecenterTimer = setTimeout(recenterManual, 6000); });
        }

        function recenterManual() {
            mapaDesplazadoPorUsuario = false;
            document.getElementById('recenter-container').style.display = 'none';
            clearTimeout(autoRecenterTimer);
            map.easeTo({ center: miPosicion, zoom: 16.8, bearing: lastValidBearing, pitch: 62, offset: getCenterOffset(), duration: 600 });
        }

        function configurarCapasRuta() {
            map.addSource('route-source', { type: 'geojson', data: { type: 'FeatureCollection', features: [] } });
            map.addLayer({ id: 'route-casing', type: 'line', source: 'route-source', layout: { 'line-join': 'round', 'line-cap': 'round' }, paint: { 'line-color': '#000000', 'line-width': 16, 'line-opacity': 0.85 } });
            map.addLayer({ id: 'route-core', type: 'line', source: 'route-source', layout: { 'line-join': 'round', 'line-cap': 'round' }, paint: { 'line-color': '#00ff66', 'line-width': 8, 'line-opacity': 0.95 } });
        }

        function crearIconoNavegacion() {
            const container = document.createElement('div');
            container.className = 'nav-arrow-marker';
            container.innerHTML = `<svg width="45" height="45" viewBox="0 0 100 100" fill="none"><path d="M50 5L90 90L50 70L10 90L50 5Z" fill="#ffffff" stroke="#00e5ff" stroke-width="6" stroke-linejoin="round"/></svg>`;
            arrowMarker = new maplibregl.Marker({ element: container }).setLngLat(miPosicion).addTo(map);
        }

        function verificarDestinoEnPerimetro(lon, lat) {
            if (!mapaMetadataActiva || !mapaMetadataActiva.bounds) return true;
            // Tolerancia de seguridad perimetral de rutas expandida
            return true; 
        }

        async function fijarRutaDestino() {
            const query = document.getElementById('destination-input').value;
            if (!query) return;
            try {
                const searchResponse = await fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(query)}&limit=1`);
                const searchData = await searchResponse.json();
                if (searchData.length === 0) return;
                
                destinoCoors = [parseFloat(searchData[0].lon), parseFloat(searchData[0].lat)];
                
                rutaActiva = true;
                document.getElementById('search-block').style.display = 'none';
                document.getElementById('hud-top-danger-meter').style.display = 'flex';
                document.getElementById('trip-totals-pill').style.display = 'flex';
                document.getElementById('panel-right-street').style.display = 'flex';
                
                actualizarRutaEnMarcha();
                recenterManual();
            } catch (error) { 
                alert("Fallo de red al buscar el destino.");
            }
        }

        async function actualizarRutaEnMarcha() {
            if (!rutaActiva || !destinoCoors) return;
            let dataRuta = null;
            const controller = new AbortController();
            const timeoutId = setTimeout(() => controller.abort(), 3000);

            try {
                const routeResponse = await fetch(`https://router.project-osrm.org/route/v1/driving/${miPosicion[0]},${miPosicion[1]};${destinoCoors[0]},${destinoCoors[1]}?overview=full&geometries=geojson&steps=true`, { signal: controller.signal });
                clearTimeout(timeoutId);
                const routeData = await routeResponse.json();
                
                if (routeData.routes && routeData.routes.length > 0) {
                    dataRuta = routeData.routes[0];
                    cacheRutaLocal = dataRuta;
                    document.getElementById('offline-alert-banner').style.display = 'none';
                }
            } catch (e) {
                clearTimeout(timeoutId);
                if (cacheRutaLocal) {
                    dataRuta = cacheRutaLocal;
                    document.getElementById('offline-alert-banner').style.display = 'block';
                } else { return; }
            }

            if (!dataRuta || !map.getSource('route-source')) return;

            map.getSource('route-source').setData({ type: 'FeatureCollection', features: [{ type: 'Feature', geometry: dataRuta.geometry, properties: {} }] });
            
            const kms = (dataRuta.distance / 1000).toFixed(1);
            document.getElementById('total-distance-display').innerHTML = `<span>${kms}</span> KM`;
            const totalMinutos = Math.round(dataRuta.duration / 60);
            if (totalMinutos >= 60) { 
                document.getElementById('total-time-display').innerHTML = `<span>${Math.floor(totalMinutos / 60)}h ${totalMinutos % 60}m</span>`; 
            } else { 
                document.getElementById('total-time-display').innerHTML = `<span>${totalMinutos}</span> MIN`; 
            }

            const steps = dataRuta.legs[0].steps;
            if (steps && steps.length > 1) {
                const metrosRealesGiro = Math.round(steps[0].distance); 
                const proximoGiro = steps[1];
                document.getElementById('next-street-name').innerText = proximoGiro.name || proximoGiro.ref || "Tramo Continuo";
                document.getElementById('hud-curve-distance').innerText = cuantificarMetrosCurva(metrosRealesGiro);

                const modifier = proximoGiro.maneuver.modifier || '';
                let direccion = 'recto';
                if (modifier.includes('left')) direccion = 'izq';
                if (modifier.includes('right')) direccion = 'der';

                let peligrosidad = 'low'; 
                const maneuverType = proximoGiro.maneuver.type;

                if (motos_danger_types_medium.includes(maneuverType) || motos_danger_modifiers_medium.includes(modifier)) peligrosidad = 'medium';
                if (motos_danger_modifiers_high.includes(modifier)) peligrosidad = 'high';

                renderizarHUDPeligro(direccion, peligrosidad);
            }
        }

        const motos_danger_types_medium = ['turn', 'on ramp'];
        const motos_danger_modifiers_medium = ['slight left', 'slight right', 'left', 'right'];
        const motos_danger_modifiers_high = ['sharp left', 'sharp right', 'uturn'];

        function cuantificarMetrosCurva(metros) {
            if (metros <= 25) return "25 m";
            if (metros <= 50) return "50 m";
            if (metros <= 100) return "100 m";
            if (metros <= 150) return "150 m";
            return (Math.round(metros / 50) * 50) + " m";
        }

        function renderizarHUDPeligro(direccion, peligrosidad) {
            const hudTop = document.getElementById('hud-top-danger-meter');
            const container = document.getElementById('danger-chevrons-container');
            container.innerHTML = ''; 

            const config = CONFIG_PELIGRO[peligrosidad] || CONFIG_PELIGRO.low;
            hudTop.style.background = config.fondo;

            const scaleX = (direccion === 'izq') ? -1 : 1;
            const svgChevron = `<svg class="danger-chevron" style="transform: scaleX(${scaleX});" viewBox="0 0 24 24" fill="none"><path d="M10 5L17 12L10 19" stroke="${config.color}" stroke-width="5" stroke-linecap="round" stroke-linejoin="round"/></svg>`;

            for (let i = 0; i < config.paneles; i++) { container.innerHTML += svgChevron; }
        }

        function cancelarRuta() { 
            rutaActiva = false; destinoCoors = null; cacheRutaLocal = null; 
            if (map.getSource('route-source')) map.getSource('route-source').setData({ type: 'FeatureCollection', features: [] }); 
            document.getElementById('hud-top-danger-meter').style.display = 'none'; 
            document.getElementById('trip-totals-pill').style.display = 'none'; 
            document.getElementById('panel-right-street').style.display = 'none'; 
            document.getElementById('offline-alert-banner').style.display = 'none'; 
            document.getElementById('perimeter-alert-banner').style.display = 'none'; 
            document.getElementById('search-block').style.display = 'flex'; 
            document.getElementById('destination-input').value = ''; 
            recenterManual(); 
        }

        function arrancarSeguimientoGPS() {
            if (!navigator.geolocation) return;
            navigator.geolocation.watchPosition((position) => {
                document.getElementById('gps-alert-banner').style.display = 'none';
                const lon = position.coords.longitude;
                const lat = position.coords.latitude;
                const velocidad = position.coords.speed ? Math.round(position.coords.speed * 3.6) : 0;
                
                miPosicion = [lon, lat];
                if (position.coords.heading !== null && position.coords.heading !== undefined) lastValidBearing = position.coords.heading;
                
                document.getElementById('speed-value').innerText = velocidad;
                if (arrowMarker) arrowMarker.setLngLat(miPosicion);
                
                const arrowElement = document.querySelector('.nav-arrow-marker');
                if (arrowElement) arrowElement.style.transform = `rotate(${lastValidBearing}deg)`;
                if (rutaActiva) actualizarRutaEnMarcha();
                
                if (!mapaDesplazadoPorUsuario) { 
                    map.easeTo({ center: miPosicion, bearing: lastValidBearing, offset: getCenterOffset(), duration: 280, easing: (t) => t }); 
                }
            }, (err) => {
                document.getElementById('gps-alert-banner').style.display = 'block';
            }, { enableHighAccuracy: true, maximumAge: 3000, timeout: 15000 });
        }
